/**
 * Kits para atención de derrames · Méderi
 * Archivo Code.gs — va en Extensiones > Apps Script de la hoja de cálculo.
 * Los datos se guardan en las hojas "Productos" y "Kits" de esta misma hoja de cálculo.
 */
 
const HOJAS = {
  productos: { nombre: 'Productos', columnas: ['id','kit','producto','vence','historial','orden','actualizado'] },
  kits:      { nombre: 'Kits',      columnas: ['id','numero','nombre'] }
};
const SESION_HORAS = 6;
 
/* ---------- Página web ---------- */
function doGet() {
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('Kits de Derrame Mederi')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}
 
/* ---------- Menú en la hoja de cálculo ---------- */
function onOpen() {
  SpreadsheetApp.getUi().createMenu('Kits de derrame')
    .addItem('1. Cargar datos iniciales', 'cargarDatosIniciales')
    .addItem('2. Crear o cambiar acceso (correo y contraseña)', 'crearAcceso')
    .addToUi();
}
 
function hoja_(col) {
  const def = HOJAS[col];
  if (!def) throw new Error('Colección no válida');
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(def.nombre);
  if (!sh) {
    sh = ss.insertSheet(def.nombre);
    sh.getRange(1, 1, 1, def.columnas.length).setValues([def.columnas]).setFontWeight('bold').setBackground('#ff671d').setFontColor('#ffffff');
    sh.setFrozenRows(1);
    sh.getRange('A:Z').setNumberFormat('@'); // todo como texto para que Sheets no cambie las fechas
  }
  return sh;
}
 
function leer_(col) {
  const def = HOJAS[col], sh = hoja_(col);
  const n = sh.getLastRow();
  if (n < 2) return [];
  const vals = sh.getRange(2, 1, n - 1, def.columnas.length).getValues();
  return vals.filter(r => r[0] !== '').map(r => {
    const o = {};
    def.columnas.forEach((c, i) => {
      let v = r[i];
      if (v instanceof Date) v = Utilities.formatDate(v, Session.getScriptTimeZone(), 'yyyy-MM-dd');
      if (c === 'historial') { try { v = v ? JSON.parse(v) : []; } catch (e) { v = []; } }
      if (c === 'kit' || c === 'numero' || c === 'orden') v = Number(v) || 0;
      o[c] = v;
    });
    return o;
  });
}
 
function fila_(col, obj) {
  return HOJAS[col].columnas.map(c => {
    const v = obj[c];
    if (c === 'historial') return JSON.stringify(v || []);
    return v === undefined || v === null ? '' : String(v);
  });
}
 
/* ---------- Lectura (abierta para todos) ---------- */
function obtenerDatos() {
  return { productos: leer_('productos'), kits: leer_('kits') };
}
 
/* ---------- Acceso ---------- */
function hash_(email, pass, salt) {
  let h = salt + '|' + String(email).trim().toLowerCase() + '|' + pass;
  for (let i = 0; i < 2000; i++) {
    h = Utilities.base64Encode(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, h, Utilities.Charset.UTF_8));
  }
  return h;
}
 
function crearAcceso() {
  const ui = SpreadsheetApp.getUi();
  const r1 = ui.prompt('Acceso de edición', 'Correo con el que se ingresará:', ui.ButtonSet.OK_CANCEL);
  if (r1.getSelectedButton() !== ui.Button.OK) return;
  const r2 = ui.prompt('Acceso de edición', 'Contraseña (mínimo 8 caracteres):', ui.ButtonSet.OK_CANCEL);
  if (r2.getSelectedButton() !== ui.Button.OK) return;
  const email = r1.getResponseText().trim().toLowerCase(), pass = r2.getResponseText();
  if (!email || pass.length < 8) { ui.alert('La contraseña debe tener al menos 8 caracteres.'); return; }
  const salt = Utilities.getUuid();
  PropertiesService.getScriptProperties().setProperties({ ACCESO_SALT: salt, ACCESO_HASH: hash_(email, pass, salt) });
  ui.alert('Acceso guardado. Ya puedes ingresar en la página con ese correo y contraseña.');
}
 
function ingresar(email, pass) {
  const props = PropertiesService.getScriptProperties();
  const salt = props.getProperty('ACCESO_SALT'), hash = props.getProperty('ACCESO_HASH');
  if (!salt || !hash) throw new Error('Aún no se ha creado el acceso. Usa el menú "Kits de derrame" en la hoja de cálculo.');
  const cache = CacheService.getScriptCache();
  const fallos = Number(cache.get('fallos') || 0);
  if (fallos >= 5) throw new Error('Demasiados intentos. Espera un minuto.');
  if (hash_(email, pass, salt) !== hash) {
    cache.put('fallos', String(fallos + 1), 60);
    throw new Error('Correo o contraseña incorrectos.');
  }
  const token = Utilities.getUuid();
  cache.put('t_' + token, String(email).trim().toLowerCase(), SESION_HORAS * 3600);
  return { token: token, email: String(email).trim().toLowerCase() };
}
 
function validar_(token) {
  const email = token && CacheService.getScriptCache().get('t_' + token);
  if (!email) throw new Error('SESION: tu sesión terminó. Ingresa de nuevo.');
  return email;
}
 
function salir(token) { if (token) CacheService.getScriptCache().remove('t_' + token); return true; }
 
/* ---------- Escritura (requiere sesión) ---------- */
function buscarFila_(sh, id) {
  const n = sh.getLastRow();
  if (n < 2) return -1;
  const ids = sh.getRange(2, 1, n - 1, 1).getValues();
  for (let i = 0; i < ids.length; i++) if (String(ids[i][0]) === String(id)) return i + 2;
  return -1;
}
 
function guardar(token, col, id, datos) {
  validar_(token);
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const sh = hoja_(col);
    id = id || Utilities.getUuid().slice(0, 20);
    const obj = Object.assign({}, datos, { id: id });
    const row = buscarFila_(sh, id);
    const valores = [fila_(col, obj)];
    if (row > 0) sh.getRange(row, 1, 1, valores[0].length).setValues(valores);
    else sh.appendRow(valores[0]);
    return id;
  } finally { lock.releaseLock(); }
}
 
function actualizar(token, col, id, cambios) {
  validar_(token);
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const sh = hoja_(col), row = buscarFila_(sh, id);
    if (row < 0) throw new Error('No se encontró el registro.');
    const actual = leer_(col).find(o => String(o.id) === String(id));
    const obj = Object.assign({}, actual, cambios, { id: id });
    const valores = [fila_(col, obj)];
    sh.getRange(row, 1, 1, valores[0].length).setValues(valores);
    return id;
  } finally { lock.releaseLock(); }
}
 
function borrar(token, col, id) {
  validar_(token);
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const sh = hoja_(col), row = buscarFila_(sh, id);
    if (row > 0) sh.deleteRow(row);
    return true;
  } finally { lock.releaseLock(); }
}

function cargarDatosIniciales() {
  const ui = SpreadsheetApp.getUi();
  const hp = hoja_('productos'), hk = hoja_('kits');
  if (hp.getLastRow() > 1 || hk.getLastRow() > 1) {
